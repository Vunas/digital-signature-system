Mở file `app/routers/document_router.py` (hoặc nơi bạn định nghĩa các route cho document) và dán thêm đoạn code sau vào cuối file: ```python from fastapi import APIRouter, Depends, HTTPException, status from fastapi.responses import FileResponse from sqlalchemy.orm import Session import os from app.core.dependencies import get_db, get_current_user from app.models.user import User from app.repositories.document_repo import document_repo # Đảm bảo router đã được khởi tạo: router = APIRouter(prefix="/documents", tags=["Documents"]) @router.get("/{document_id}/download") def download_document( document_id: int, db: Session = Depends(get_db), current_user: User = Depends(get_current_user) ): """ API Tải tài liệu về máy. - Nếu tài liệu đã được ký, trả về bản đã ký (có nhúng PAdES). - Nếu chưa ký, trả về bản nháp gốc. """ doc = document_repo.get_by_id(db, document_id, current_user.id) if not doc: raise HTTPException(status_code=404, detail="Không tìm thấy tài liệu") # Ưu tiên tải bản đã ký (nếu có), nếu chưa ký thì tải bản gốc file_path = doc.signed_file_path if doc.status == 'SIGNED' else doc.original_file_path if not file_path or not os.path.exists(file_path): raise HTTPException(status_code=404, detail="File vật lý không tồn tại trên hệ thống") # Xác định tên file tải về (thêm đuôi _signed để dễ nhận diện nếu đã ký) download_filename = doc.file_name if doc.status == 'SIGNED' and not download_filename.endswith('_signed.pdf'): download_filename = download_filename.replace('.pdf', '_signed.pdf') # Dùng FileResponse của FastAPI để tự động streaming file về cho trình duyệt return FileResponse( path=file_path, filename=download_filename, media_type="application/pdf" )
